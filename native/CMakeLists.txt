cmake_minimum_required(VERSION 3.20)
project(cpzero_native LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_CLI_STATIC OFF CACHE BOOL "" FORCE)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_INCLUDE_SIMPLE ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(CPZERO_BACKEND "SDL" CACHE STRING "Native output backend: SDL or FBDEV")
set_property(CACHE CPZERO_BACKEND PROPERTY STRINGS SDL FBDEV)
if(NOT CPZERO_BACKEND STREQUAL "SDL" AND NOT CPZERO_BACKEND STREQUAL "FBDEV")
  message(FATAL_ERROR "CPZERO_BACKEND must be SDL or FBDEV")
endif()
if(CPZERO_BACKEND STREQUAL "FBDEV" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "CPZERO_BACKEND=FBDEV requires Linux")
endif()
FetchContent_Declare(lvgl URL https://codeload.github.com/lvgl/lvgl/tar.gz/7f07a129e8d77f4984fff8e623fd5be18ff42e74 URL_HASH SHA256=8a5e30223fd5b08cdeccb060f45b37a29868d6b1fdfc5bdbdf0dd89aaefb5a0e DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(quickjs URL https://codeload.github.com/quickjs-ng/quickjs/tar.gz/670492dd342dace0bb7bd6fbfbde8f0bc5651224 URL_HASH SHA256=4c875fae3c1df5764d135898c2d4bef4501a34d018cfac466c46bd61fa888def DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(lvgl quickjs)
add_executable(cpzero-host host.cpp)
set(CPZERO_EXTENSION_SOURCES "" CACHE STRING "Semicolon-separated native service adapter source files")
if(CPZERO_EXTENSION_SOURCES)
  target_sources(cpzero-host PRIVATE ${CPZERO_EXTENSION_SOURCES})
endif()
target_include_directories(cpzero-host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${quickjs_SOURCE_DIR})
target_link_libraries(cpzero-host PRIVATE lvgl::lvgl qjs)
if(CPZERO_BACKEND STREQUAL "SDL")
  find_package(PkgConfig QUIET)
  if(APPLE)
    find_package(SDL2 CONFIG QUIET)
  endif()
  if(TARGET SDL2::SDL2)
    set(CPZERO_SDL SDL2::SDL2)
  else()
    if(NOT PkgConfig_FOUND)
      message(FATAL_ERROR "SDL2 development files are required (macOS: brew install sdl2; Linux: install libsdl2-dev)")
    endif()
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    set(CPZERO_SDL PkgConfig::SDL2)
  endif()
  target_link_libraries(cpzero-host PRIVATE ${CPZERO_SDL})
else()
  target_compile_definitions(cpzero-host PRIVATE CPZERO_FBDEV=1)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(cpzero-host PRIVATE m pthread dl)
endif()
